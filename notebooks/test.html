ojsBridge = (id, value) => {
  const key = `ojs_${id}`;
  const eventName = `ojs:${id}:change`;

  // Publish the current OJS value.
  window[key] = value;
  window.dispatchEvent(new CustomEvent(eventName, { detail: value }));

  // Receive JavaScript updates as an observable.
  return Generators.observe((notify) => {
    const handler = (e) => notify(e.detail);
    window.addEventListener(eventName, handler);

    if (key in window) notify(window[key]);

    return () => window.removeEventListener(eventName, handler);
  });
};

The following checks exercise the DOM contract and commit timing without requiring visual inspection. Commit timing is fixed: selecting an option must not emit, and closing the menu emits once if anything changed while it was open.

multiSelectBehaviorChecks = {
  const results = [];
  const check = (label, pass) => results.push({ label, pass: Boolean(pass) });
  const sandbox = document.createElement("div");
  sandbox.style.cssText = "position:fixed;left:-10000px;top:0;width:500px;";
  document.body.appendChild(sandbox);

  const immediate = atlasMultiSelect(["Alpha", "Beta"], {
    value: [],
    requireAtLeastOne: false,
    enableSelectAll: true,
  });
  sandbox.appendChild(immediate);
  let immediateEvents = 0;
  immediate.addEventListener("input", () => immediateEvents += 1);
  const immediateButton = immediate.querySelector("button");
  immediateButton.click();
  const immediateOption = immediate.querySelector('[role="option"]');
  immediateOption.click();

  check("selecting an option does not emit", immediateEvents === 0);
  check("trigger exposes expanded state", immediateButton.getAttribute("aria-expanded") === "true");
  check("options use listbox semantics", immediate.querySelector('[role="listbox"]')?.contains(immediateOption));
  check("all buttons have type=button", Array.from(immediate.querySelectorAll("button")).every((button) => button.type === "button"));

  immediateOption.dispatchEvent(new KeyboardEvent("keydown", {
    key: "ArrowDown",
    bubbles: true,
  }));
  check("ArrowDown moves option focus", document.activeElement === immediate.querySelectorAll('[role="option"]')[1]);

  document.activeElement.dispatchEvent(new KeyboardEvent("keydown", {
    key: "Escape",
    bubbles: true,
  }));
  check("Escape closes the menu", immediateButton.getAttribute("aria-expanded") === "false");
  check("Escape returns trigger focus", document.activeElement === immediateButton);
  // Closing is the commit, however it is closed.
  check("Escape commits the pending change", immediateEvents === 1);

  const deferred = atlasMultiSelect(["Alpha", "Beta"], {
    value: [],
    requireAtLeastOne: false,
  });
  sandbox.appendChild(deferred);
  let deferredEvents = 0;
  deferred.addEventListener("input", () => deferredEvents += 1);
  deferred.atlasMultiSelect.open();
  deferred.querySelector('[role="option"]').click();
  check("selection while open is deferred", deferredEvents === 0);
  deferred.atlasMultiSelect.close();
  check("close emits exactly once", deferredEvents === 1);
  deferred.atlasMultiSelect.open();
  deferred.atlasMultiSelect.close();
  check("close with no change emits nothing", deferredEvents === 1);

  const localized = atlasMultiSelect(["Un", "Deux"], {
    value: [],
    requireAtLeastOne: false,
    enableSelectAll: true,
    language: "fr",
  });
  sandbox.appendChild(localized);
  check("empty label is localized", localized.querySelector("button span")?.textContent === "Aucune sélection");
  check("action labels are localized", Array.from(localized.querySelectorAll(".button-container button"), (button) => button.textContent).join("|") === "Tout sélectionner|Tout désélectionner");

  const limited = atlasMultiSelect(["One", "Two"], {
    value: [],
    maxSelections: 1,
    requireAtLeastOne: false,
    enableSelectAll: true,
  });
  sandbox.appendChild(limited);
  const limitedActions = Array.from(
    limited.querySelectorAll(".button-container button"),
    (button) => button.textContent,
  );
  check("Select All is hidden when selections are limited", !limitedActions.includes("Select All"));
  check("valid bulk actions remain available", limitedActions.join("|") === "Deselect All");

  const nativeSelect = deferred.querySelector("select");
  const addedOption = document.createElement("option");
  addedOption.value = "Gamma";
  addedOption.textContent = "Gamma <safe>";
  nativeSelect.appendChild(addedOption);
  await new Promise((resolve) => window.setTimeout(resolve, 0));
  check("option mutations rebuild the list", deferred.querySelectorAll('[role="option"]').length === 3);
  const addedRow = deferred.querySelectorAll('[role="option"]')[2];
  check("option labels remain text", addedRow?.textContent.includes("Gamma <safe>") && !addedRow.querySelector("*:not(span)"));

  [immediate, deferred, localized, limited].forEach((input) => {
    input.atlasMultiSelect.destroy();
  });
  sandbox.remove();

  const list = document.createElement("ul");
  results.forEach(({ label, pass }) => {
    const item = document.createElement("li");
    item.textContent = `${pass ? "PASS" : "FAIL"} — ${label}`;
    item.style.color = pass
      ? "var(--atlas-color-primary)"
      : "var(--atlas-color-focus)";
    list.appendChild(item);
  });
  return list;
}

Canonical administrative selector checks

The shared selector is helpers/adminSelectors.ojs. The selection lives in that module, one cell per level, and every control rendered anywhere on the page binds to those same cells. Check that:

  • changing a parent removes child selections that no longer belong to it — and that it does so for a level with no control on screen (the partial form below shows only country and region; pick a district in the full form first, then change the country in the partial one);
  • the partial form and the full form stay in sync, and using the partial one never clears the district selection by itself;
  • Côte d’Ivoire and names containing apostrophes work without affecting queries;
  • region and district search works with or without accents;
  • Select All is omitted when a level has a finite maximum;
  • controls wrap cleanly in a narrow window;
  • each trigger label reports only what is actually selected. Select All on countries, then check that regions reads “All regions” rather than a count of countries.

One whole-area convention throughout: a row whose own name is null means the level is not narrowed. Single-select shows those rows (“Sub-Saharan Africa”, “Whole country”, “All districts”); multi-select expresses the same thing by selecting nothing, so it has no such rows.

import {
  adminSelect,
  adminMultiSelect,
  adminSelections,
} from "/helpers/adminSelectors.ojs";

adminSelectorSummary = (selection) =>
  Object.fromEntries(
    Object.entries(selection).map(([level, rows]) => [
      level,
      // Empty means the level is not narrowed, in either mode.
      rows.map((row) =>
        [row.iso3, row.admin1_name, row.admin2_name]
          .filter((part) => part != null)
          .join(" / "),
      ),
    ]),
  );
