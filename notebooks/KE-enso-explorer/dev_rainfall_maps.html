md`### Notes
- **Rainfall data:** per-year seasonal PTOT COGs (D409 obs pipeline, \`processing=seasonal\`,
  \`PTOT_{SEASON}_{YYYY}_sum.tif\`), one fetch per panel; monthly-sum fallback if a
  year is missing. Anomaly subtracts the 1991–2020 climatology COG. 2026 excluded.
- **Drought data (SPEI-3):** monthly per-pixel COGs (\`processing=monthly/variable=SPEI-03\`,
  \`SPEI-03-{YYYY}-{MM}.tif\`), read at the season-end month (OND → Dec, MAM → May) — a
  3-month accumulation already covers the season. SPEI is already a standardised anomaly
  (−ve drier, +ve wetter), so the anomaly toggle is disabled for it.
- **Vegetation data (NDVI):** MODIS **MOD13Q1** v061 per-year seasonal-mean COGs (250 m,
  \`type=vegetation/source=modis-mod13q1/region=east-africa\`), real NDVI (DN/10000, ~0–1),
  pixel-reliability masked. Its own 250 m East-Africa grid (≠ the ~5 km CHIRPS grid) so the
  read window + county mask recompute for it. Anomaly toggle disabled (no NDVI climatology COG yet).
- **Flood data (observed):** Copernicus **GFM** (Sentinel-1 SAR, ~111 m) seasonal 0/1 occurrence COGs
  (\`type=flood/source=glofas-gfm/region=kenya/processing=seasonal/variable=flooded/season={S}/flooded_{S}_{YYYY}.tif\`),
  2018–2025 — replaced the old GFD/MODIS. Coding 0/1/**255=not-observed** (SAR gap → mapped to NaN, not
  dry). Card mean = flooded share of the SAR-observed county area. Short record = recent flood-proneness,
  NOT return-period magnitude (use JRC below for that); don't splice with the old GFD (methodological
  break). Own ~111 m grid. (History \`frequency\`/\`footprint\` COGs also live — candidate static layer.)
- **Flood hazard (JRC GloFAS):** static return-period flood **depth (m)** COGs (90 m,
  \`type=flood/source=jrc-glofas/…/rp={RP}/flood-depth_rp{RP}.tif\`), one map per return period
  (10–500 yr) via the RP slider. **Static** — same every year, independent of ENSO; it answers
  "where is flood-prone + how deep", read alongside the observed-flood years. Own 90 m grid.
- **Correlation:** Pearson r between the county-mean seasonal value (rainfall *or* SPEI-3)
  and each driver's standardised index over the selected years (n shown). Higher |r| = that
  driver explains more of this county's outcome that season; +r → driver-high raises the outcome
  (wetter / greener / better crop-water / more flood, per the selected variable).
- **Filter:** min/max on the section driver's value (raw index for single drivers,
  composite z for ENSO+IOD). Card colour + phase still standardised (z vs 1991–2020).
- **Deferred:** multi-county maps (#3); NDVI annual composite + anomaly-vs-climatology (v1 = seasonal
  OND/MAM only); NPP/biomass (superseded by NDVI); WRSI; riverine flood — see dispatches 2026-08.`

  return [xmin - px, ymin - py, xmax + px, ymax + py];

gridWindow = {
  const gt = await geotiffLib;
  const tiff = await gt.fromUrl(gridRefURL);
  const image = await tiff.getImage();
  const [ox0, oy0, ox1, oy1] = image.getBoundingBox();
  const W = image.getWidth(), H = image.getHeight();
  const xres = (ox1 - ox0) / W, yres = (oy1 - oy0) / H;
  const [bx0, by0, bx1, by1] = countyBbox;
  const px0 = Math.max(0, Math.floor((bx0 - ox0) / xres));
  const px1 = Math.min(W, Math.ceil((bx1 - ox0) / xres));
  const py0 = Math.max(0, Math.floor((oy1 - by1) / yres));
  const py1 = Math.min(H, Math.ceil((oy1 - by0) / yres));
  return {
    window: [px0, py0, px1, py1], w: px1 - px0, h: py1 - py0,
    bbox: [ox0 + px0 * xres, oy1 - py1 * yres, ox0 + px1 * xres, oy1 - py0 * yres],
  };
}
countyMask = {
  const win = await gridWindow;
  const [bx0, by0, bx1, by1] = win.bbox, w = win.w, h = win.h;
  const rings = ringsOf(countyA1);
  const inside = (x, y) => {
    let c = false;
    for (const ring of rings)
      for (let i = 0, j = ring.length - 1; i < ring.length; j = i++) {
        const xi = ring[i][0], yi = ring[i][1], xj = ring[j][0], yj = ring[j][1];
        if (((yi > y) !== (yj > y)) && (x < (xj - xi) * (y - yi) / (yj - yi) + xi)) c = !c;
      }
    return c;
  };
  const mask = new Uint8Array(w * h);
  for (let j = 0; j < h; j++) {
    const lat = by1 - (j + 0.5) / h * (by1 - by0);
    for (let i = 0; i < w; i++) {
      const lon = bx0 + (i + 0.5) / w * (bx1 - bx0);
      mask[j * w + i] = inside(lon, lat) ? 1 : 0;
    }
  }
  return mask;
}
readWindow = async (url, win) => {
  const gt = await geotiffLib;
  const tiff = await gt.fromUrl(url);
  const image = await tiff.getImage();
  const rr = await image.readRasters({window: win.window});
  const src = rr[0], out = new Float32Array(src.length);
  for (let i = 0; i < src.length; i++) {
    const v = src[i];
    out[i] = (v == null || !isFinite(v) || v <= -9999) ? NaN : v;
  }
  return out;
}

monthlySum = async (season, year, win) => {
  const mons = seasonMonthsFor[season];
  const grids = await Promise.all(mons.map((m) => loadMonthly(year, m, win)));
  const n = win.w * win.h, out = new Float32Array(n);
  for (let i = 0; i < n; i++) {
    let s = 0, ok = true;
    for (const g of grids) { const v = g[i]; if (Number.isNaN(v)) { ok = false; break; } s += v; }
    out[i] = ok ? s : NaN;
  }
  return out;
}
seasonalYear = async (season, year, win) => {
  try {
    const d = await readWindow(seasonalURL(season, year), win);
    let mx = -Infinity, any = false;
    for (const v of d) if (!Number.isNaN(v)) { any = true; if (v > mx) mx = v; }
    if (any && mx > 0) return d; // valid seasonal COG
  } catch (e) { /* fall through */ }
  return monthlySum(season, year, win); // degenerate/missing → monthly sum
}

// DERIVE: apply the anomaly subtraction (if any) + compute the county mean. No fetch
// of the year stack — reuses the cached raw arrays, so the anomaly toggle is instant.
deriveCache = async (season) => {
  const raw = await seasonRawP.p, win = await gridWindow, mask = await countyMask;
  const clim = climForSeason, n = win.w * win.h;
  return raw.map(({year, data}) => {
    if (data == null) return {year, data: null, meanmm: null};
    let d = data;
    if (clim) {
      const o = new Float32Array(n);
      for (let i = 0; i < n; i++) o[i] = (Number.isNaN(data[i]) || Number.isNaN(clim[i])) ? NaN : data[i] - clim[i];
      d = o;
    }
    if (isFlood) {
      // GFM: flooded share of the SAR-OBSERVED county area = flooded / observed(0|1).
      // 255 (not observed) is already NaN → excluded from BOTH numerator + denominator
      // (it is a SAR gap, not dry). null when the county had no observation that season.
      let flooded = 0, observed = 0;
      for (let i = 0; i < n; i++) {
        if (!mask[i]) continue;
        const v = d[i];
        if (!Number.isNaN(v) && v > -9999) { observed++; if (v > 0) flooded++; }
      }
      return {year, data: d, meanmm: observed ? flooded / observed : null};
    }
    let sum = 0, cnt = 0;
    for (let i = 0; i < n; i++) { const v = d[i]; if (mask[i] && !Number.isNaN(v) && v > -9999) { sum += v; cnt++; } }
    return {year, data: d, meanmm: cnt ? sum / cnt : null};
  });
}

pearson = (pairs) => {
  const n = pairs.length; if (n < 3) return null;
  const mx = d3.mean(pairs, (p) => p.x), my = d3.mean(pairs, (p) => p.y);
  const cov = d3.sum(pairs, (p) => (p.x - mx) * (p.y - my));
  const sx = Math.sqrt(d3.sum(pairs, (p) => (p.x - mx) ** 2));
  const sy = Math.sqrt(d3.sum(pairs, (p) => (p.y - my) ** 2));
  return (sx && sy) ? cov / (sx * sy) : null;
}
// r between two aligned arrays (nulls dropped pairwise)
corrArr = (xs, ys) => pearson(xs.map((x, i) => ({x, y: ys[i]})).filter((p) => p.x != null && !Number.isNaN(p.x) && p.y != null))
// ---- small OLS + partial-correlation engine (evidence-based combination) ----
solveLin = (M, v) => {
  const n = v.length, A = M.map((r, i) => [...r, v[i]]);
  for (let c = 0; c < n; c++) {
    let piv = c;
    for (let r = c + 1; r < n; r++) if (Math.abs(A[r][c]) > Math.abs(A[piv][c])) piv = r;
    if (Math.abs(A[piv][c]) < 1e-12) return null;
    [A[c], A[piv]] = [A[piv], A[c]];
    const d = A[c][c];
    for (let j = c; j <= n; j++) A[c][j] /= d;
    for (let r = 0; r < n; r++) if (r !== c) { const fr = A[r][c]; for (let j = c; j <= n; j++) A[r][j] -= fr * A[c][j]; }
  }
  return A.map((r) => r[n]);
}
// OLS of y on intercept + cols; returns {r2, predicted}
ols = (cols, y) => {
  const n = y.length, k = cols.length + 1;
  const A = [];
  for (let i = 0; i < n; i++) { const row = [1]; for (const c of cols) row.push(c[i]); A.push(row); }
  const AtA = Array.from({length: k}, () => new Array(k).fill(0)), Aty = new Array(k).fill(0);
  for (let i = 0; i < n; i++) for (let a = 0; a < k; a++) { Aty[a] += A[i][a] * y[i]; for (let b = 0; b < k; b++) AtA[a][b] += A[i][a] * A[i][b]; }
  const bcoef = solveLin(AtA, Aty);
  if (!bcoef) return null;
  const pred = A.map((row) => row.reduce((s, v, j) => s + v * bcoef[j], 0));
  const my = d3.mean(y), sstot = d3.sum(y.map((v) => (v - my) ** 2)), ssres = d3.sum(y.map((v, i) => (v - pred[i]) ** 2));
  return {r2: sstot > 0 ? 1 - ssres / sstot : null, predicted: pred};
}
// partial r of xk with y, controlling for the `others` predictors (residual method)
partialR = (xk, others, y) => {
  if (!others.length) return corrArr(xk, y);
  const fx = ols(others, xk), fy = ols(others, y);
  if (!fx || !fy) return null;
  return corrArr(xk.map((v, i) => v - fx.predicted[i]), y.map((v, i) => v - fy.predicted[i]));
}
shortName = (k) => k.replace(/ \(.*\)$/, "")
// evidence-based driver↔rainfall stats for a set of drivers: bivariate r,
// partial r (each driver controlling for the others), best-linear multiple R
// (adjusted for df), and predictor collinearity.
corrStats = (season, cache, keys) => {
  const zmaps = keys.map((k) => zSeries(k, season));
  const complete = cache.filter((c) => c.meanmm != null)
    .map((c) => ({y: c.meanmm, xs: zmaps.map((z) => z.get(c.year))}))
    .filter((r) => r.xs.every((v) => v != null && !Number.isNaN(v)));
  const n = complete.length;
  const y = complete.map((r) => r.y);
  const cols = keys.map((_, ki) => complete.map((r) => r.xs[ki]));
  const per = keys.map((k, ki) => ({
    key: k, name: shortName(k),
    r: corrArr(cols[ki], y),
    partial: partialR(cols[ki], cols.filter((_, j) => j !== ki), y),
  }));
  const full = n > keys.length + 1 ? ols(cols, y) : null;
  const R2 = full && full.r2 != null ? Math.max(0, full.r2) : null;
  const multiR = R2 != null ? Math.sqrt(R2) : null;
  const adjR2 = R2 != null && n > keys.length + 1 ? 1 - (1 - R2) * (n - 1) / (n - keys.length - 1) : null;
  const adjR = adjR2 != null ? Math.sqrt(Math.max(0, adjR2)) : null;
  let maxColl = 0, collPair = "";
  for (let a = 0; a < keys.length; a++) for (let b = a + 1; b < keys.length; b++) {
    const rr = Math.abs(corrArr(cols[a], cols[b]) ?? 0);
    if (rr > maxColl) { maxColl = rr; collPair = `${shortName(keys[a])}↔${shortName(keys[b])}`; }
  }
  return {per, multiR, adjR, n, maxColl, collPair, k: keys.length};
}

  const meanTxt = meanmm == null ? (isFlood ? "no SAR observation" : "—")
    : isWrsi ? `${Math.round(meanmm)}% WRSI`
    : isFlood ? `${(meanmm * 100).toFixed(1)}% flooded`
    : isNdvi ? meanmm.toFixed(2)
    : isSpei ? `${meanmm > 0 ? "+" : ""}${meanmm.toFixed(2)}`
    : `${anomaly && meanmm > 0 ? "+" : ""}${Math.round(meanmm)} mm`;
  return html`<div class="rain-panel" style="background:${cardColor(spec.signal.z)}">
    <div class="rain-title"><strong>${spec.year}</strong>
      <span class="rain-mean">${meanTxt}</span>
      <div class="rain-signal">${spec.label}</div></div>
    <div class="rain-stack">${canvas}${svg}</div>
  </div>`;
}
jrcWindow = async (rp) => {
  const gt = await geotiffLib;
  const tiff = await gt.fromUrl(jrcURL(rp));
  const image = await tiff.getImage();
  const [ox0, oy0, ox1, oy1] = image.getBoundingBox();
  const W = image.getWidth(), H = image.getHeight();
  const xres = (ox1 - ox0) / W, yres = (oy1 - oy0) / H;
  const [bx0, by0, bx1, by1] = countyBbox;
  const px0 = Math.max(0, Math.floor((bx0 - ox0) / xres));
  const px1 = Math.min(W, Math.ceil((bx1 - ox0) / xres));
  const py0 = Math.max(0, Math.floor((oy1 - by1) / yres));
  const py1 = Math.min(H, Math.ceil((oy1 - by0) / yres));
  const win = {window: [px0, py0, px1, py1], w: px1 - px0, h: py1 - py0,
    bbox: [ox0 + px0 * xres, oy1 - py1 * yres, ox0 + px1 * xres, oy1 - py0 * yres]};
  const rr = await image.readRasters({window: win.window});
  const src = rr[0], n = win.w * win.h, data = new Float32Array(n);
  for (let i = 0; i < n; i++) { const v = src[i]; data[i] = (v == null || !isFinite(v) || v <= -9999) ? NaN : v; }
  // county mask for this grid + depth stats over flood-prone pixels (depth > 0)
  const [mbx0, mby0, mbx1, mby1] = win.bbox, rings = ringsOf(countyA1);
  const inside = (x, y) => { let c = false;
    for (const ring of rings) for (let i = 0, j = ring.length - 1; i < ring.length; j = i++) {
      const xi = ring[i][0], yi = ring[i][1], xj = ring[j][0], yj = ring[j][1];
      if (((yi > y) !== (yj > y)) && (x < (xj - xi) * (y - yi) / (yj - yi) + xi)) c = !c;
    } return c; };
  let inCounty = 0, floodCells = 0, sum = 0, mx = 0;
  for (let j = 0; j < win.h; j++) { const lat = mby1 - (j + 0.5) / win.h * (mby1 - mby0);
    for (let i = 0; i < win.w; i++) { const lon = mbx0 + (i + 0.5) / win.w * (mbx1 - mbx0);
      if (!inside(lon, lat)) continue; inCounty++;
      const v = data[j * win.w + i];
      if (!Number.isNaN(v) && v > 0) { floodCells++; sum += v; if (v > mx) mx = v; }
    } }
  return {...win, data, rp, share: inCounty ? floodCells / inCounty : 0,
    meanDepth: floodCells ? sum / floodCells : null, maxDepth: floodCells ? mx : null};
}
jrcDepthColor = (v) => {
  if (v == null || !isFinite(v) || v <= 0) return null; // no-flood / nodata → transparent
  const t = Math.max(0, Math.min(1, v / 4)); // depth (m), cap 4 m for contrast
  return d3.interpolateBlues(0.2 + 0.8 * t);
}
jrcRender = (win) => {
  const {data, w, h, bbox} = win, [bx0, by0, bx1, by1] = bbox;
  const aspect = ((bx1 - bx0) / (by1 - by0)) * Math.cos((by0 + by1) / 2 * Math.PI / 180);
  const W = 560, H = Math.max(220, Math.min(720, Math.round(W / aspect)));
  const tx = (lon) => (lon - bx0) / (bx1 - bx0) * W, ty = (lat) => (by1 - lat) / (by1 - by0) * H;
  const colX = new Int32Array(w + 1), rowY = new Int32Array(h + 1);
  for (let i = 0; i <= w; i++) colX[i] = Math.round(i / w * W);
  for (let j = 0; j <= h; j++) rowY[j] = Math.round(j / h * H);
  const canvas = html`<canvas width=${W} height=${H}></canvas>`;
  const ctx = canvas.getContext("2d");
  ctx.imageSmoothingEnabled = false;
  const clip = new Path2D();
  for (const ring of ringsOf(countyA1)) {
    ring.forEach(([lon, lat], i) => i ? clip.lineTo(tx(lon), ty(lat)) : clip.moveTo(tx(lon), ty(lat)));
    clip.closePath();
  }
  ctx.save(); ctx.clip(clip);
  for (let j = 0; j < h; j++) for (let i = 0; i < w; i++) {
    const c = jrcDepthColor(data[j * w + i]); if (!c) continue;
    ctx.fillStyle = c; ctx.fillRect(colX[i], rowY[j], colX[i + 1] - colX[i], rowY[j + 1] - rowY[j]);
  }
  ctx.restore();
  const ringPath = (ring) => "M" + ring.map(([lon, lat]) => `${tx(lon).toFixed(1)},${ty(lat).toFixed(1)}`).join("L") + "Z";
  const subPaths = countySubs.flatMap((f) => ringsOf(f)).map(ringPath).join(" ");
  const cntyPaths = ringsOf(countyA1).map(ringPath).join(" ");
  const svg = html`<svg viewBox="0 0 ${W} ${H}" preserveAspectRatio="none" style="position:absolute;left:0;top:0;width:100%;height:100%">
    <path d="${subPaths}" fill="none" stroke="#555" stroke-width="0.6" opacity="0.8"/>
    <path d="${cntyPaths}" fill="none" stroke="#111" stroke-width="1.4"/></svg>`;
  const rampW = 120;
  const ramp = html`<canvas width=${rampW} height=${14} style="vertical-align:middle"></canvas>`;
  const rctx = ramp.getContext("2d");
  for (let x = 0; x < rampW; x++) { const c = jrcDepthColor((x / (rampW - 1)) * 4 + 0.001); rctx.fillStyle = c || "#fff"; rctx.fillRect(x, 0, 1, 14); }
  const stat = win.meanDepth == null
    ? html`<b>no modelled river flood</b> in ${county} at this return period`
    : html`flood-prone area <b>${(win.share * 100).toFixed(1)}%</b> of county · mean depth <b>${win.meanDepth.toFixed(2)} m</b> · max <b>${win.maxDepth.toFixed(1)} m</b>`;
  return html`<div>
    <div class="jrc-stack" style="max-width:${W}px">${canvas}${svg}</div>
    <div style="font-size:0.8em;color:#555;display:flex;align-items:center;gap:0.5rem;flex-wrap:wrap;margin-top:0.35rem">
      <span>Flood depth (m), 1-in-${win.rp}-yr:</span><span>0</span>${ramp}<span>≥4</span>
      <span style="margin-left:0.6rem">${stat}</span>
    </div></div>`;
}
